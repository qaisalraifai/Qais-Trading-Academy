## 2) أضيفي الـ handlers (بعد دوال openFiles/closeFiles وقبل الـ JSX return)

دوري على `function closeFiles() {` وبعد نهايتها ضيفي:

```js
async function openAssignments(batch) {
  setAssignmentsBatch(batch);
  setAssignmentsLoading(true);
  const res = await fetch(`/api/admin/batches/${batch.id}/assignments`);
  const data = await res.json().catch(() => ({}));
  setBatchAssignments(res.ok ? data.assignments || [] : []);
  setAssignmentsLoading(false);
}

function closeAssignments() {
  setAssignmentsBatch(null);
  setBatchAssignments([]);
  setNewAssignment({ title: "", description: "", due_date: "" });
  setAssignmentError("");
}

async function handleCreateAssignment() {
  if (!newAssignment.title.trim()) {
    setAssignmentError("لازم تكتبي عنوان الواجب");
    return;
  }
  setAssignmentSaving(true);
  setAssignmentError("");
  const res = await fetch(`/api/admin/batches/${assignmentsBatch.id}/assignments`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(newAssignment),
  });
  const data = await res.json().catch(() => ({}));
  setAssignmentSaving(false);
  if (!res.ok) {
    setAssignmentError(data.error || "صار خطأ");
    return;
  }
  setBatchAssignments((prev) => [data.assignment, ...prev]);
  setNewAssignment({ title: "", description: "", due_date: "" });
}

async function handleDeleteAssignment(assignment) {
  if (!confirm(`متأكدة تبين تحذفي واجب "${assignment.title}"؟`)) return;
  const res = await fetch(`/api/admin/batches/${assignmentsBatch.id}/assignments/${assignment.id}`, {
    method: "DELETE",
  });
  if (res.ok) setBatchAssignments((prev) => prev.filter((a) => a.id !== assignment.id));
}

async function openGrading(assignment) {
  setSubmissionsLoading(true);
  const res = await fetch(`/api/admin/assignments/${assignment.id}/submissions`);
  const data = await res.json().catch(() => ({}));
  setGradingSubmission({ assignment, submissions: res.ok ? data.submissions || [] : [] });
  setSubmissionsLoading(false);
}

function closeGrading() {
  setGradingSubmission(null);
  setGradeInputs({});
}

async function handleGrade(submission) {
  const input = gradeInputs[submission.id];
  if (!input?.grade?.trim()) return;
  const res = await fetch(
    `/api/admin/assignments/${gradingSubmission.assignment.id}/submissions/${submission.id}`,
    {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ grade: input.grade.trim(), feedback: input.feedback?.trim() || "" }),
    }
  );
  const data = await res.json().catch(() => ({}));
  if (res.ok) {
    setGradingSubmission((prev) => ({
      ...prev,
      submissions: prev.submissions.map((s) => (s.id === submission.id ? data.submission : s)),
    }));
  }
}
```
