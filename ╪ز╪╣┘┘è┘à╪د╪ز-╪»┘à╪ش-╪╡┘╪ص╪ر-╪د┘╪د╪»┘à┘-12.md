  const [chatLoading, setChatLoading] = useState(false);
  const [chatText, setChatText] = useState("");
  const [chatSending, setChatSending] = useState(false);

## 2) الدوال (Handlers)

ابحثي عن نهاية قسم مكتبة الملفات — هذا السطر بالضبط:
```js
  function formatFileSize(bytes) {
    if (!bytes) return "—";
    if (bytes < 1024 * 1024) return `${Math.round(bytes / 1024)} كيلوبايت`;
    return `${(bytes / (1024 * 1024)).toFixed(1)} ميجابايت`;
  }
  // ---------------------------------------------------------------------
```

وضيفي تحته مباشرة:
```js

  // -------------------- المرحلة 12: الدردشة --------------------
  async function fetchChatMessages(batchId) {
    const res = await fetch(`/api/admin/batches/${batchId}/chat`);
    const data = await res.json();
    if (res.ok) setChatMessages(data.messages || []);
  }

  async function openChat(batch) {
    setChatBatch(batch);
    setChatError("");
    setChatLoading(true);
    await fetchChatMessages(batch.id);
    setChatLoading(false);
  }

  function closeChat() {
    setChatBatch(null);
    setChatMessages([]);
  }

  useEffect(() => {
    if (!chatBatch) return;
    const interval = setInterval(() => fetchChatMessages(chatBatch.id), 3000);
    return () => clearInterval(interval);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [chatBatch]);

  async function handleSendChat(e) {
    e.preventDefault();
    const text = chatText.trim();
    if (!text || !chatBatch) return;
    setChatSending(true);
    setChatError("");

    const res = await fetch(`/api/admin/batches/${chatBatch.id}/chat`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ message: text }),
    });
    const data = await res.json();
    setChatSending(false);

    if (!res.ok) {
      setChatError(data.error || "صار خطأ بإرسال الرسالة");
      return;
    }
    setChatText("");
    setChatMessages((prev) => [...prev, data.message]);
  }
  // -----------------------------------------------------------------
```
